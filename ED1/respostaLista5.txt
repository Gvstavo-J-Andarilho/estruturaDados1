#include <stdio.h>
// Lê os N inteiros usando aritmética de ponteiros
void maior_menor(float *v, int n,
               float *maior, int *idx_maior,
               float *menor, int *idx_menor){
    *maior = *(v+0);
    *idx_maior = 0;
    *menor = *(v+0);
    *idx_menor = 0;
    
    for(int i = 1;i<n;i++){
        if(*(v+i) > *maior){
            *maior = *(v+i);
            *idx_maior = i;
            
        }if(*(v+i) < *menor){
            *menor = *(v+i);
            *idx_menor = i;
        }
    }
}

#include <stdio.h>
// Lê os N inteiros usando aritmética de ponteiros
void soma_media(int *v, int n,
              int *soma, float *media, int *acima){
    *(soma) =0;
    for(int i = 0; i<n; i++){
        *soma += *(v+i);
    }

    if(n > 0){
    *media = (float)(*soma)/n; //transformar soma(int) em float
    }else{
        *media = 0;
    }
    *acima = 0;
    for(int i = 0; i<n;i++){
        if(*(v+i)>*media){
            (*acima)++;
        }
    }
}

int busca_linear(int *v, int n, int alvo, int *cmps){
    
    for(int i = 0; i<n; i++){
        (*cmps)++;
        if(*(v+i)==alvo){
            return i;
        }
    }
    return -1;
}

#include <stdio.h>
// Lê os N inteiros usando aritmética de ponteiros
int busca_binaria(int *v, int n, int alvo, int *cmps){
    int ini=0, fim=(n-1);
    while(ini<=fim){
        int meio=ini+(fim-ini)/2;
        (*cmps)++;
        if(*(v+meio)==alvo){
            return meio;
        } else if(*(v+meio)<alvo){
            ini = (meio+1);   
        }else{
            fim = (meio-1);
    }
    return -1;
}

int main() {
    int v[5] = {10, 25, 30, 45, 50};
    int total_cmps;
    int alvo = 45;
    int indice = busca_linear(v, 5, alvo, &total_cmps);
    if (indice != -1) {
        printf("Achei o numero %d no indice %d.\n", alvo, indice);
    } else {
        printf("Numero nao encontrado.\n");
    }
    printf("Total de comparacoes feitas: %d\n", total_cmps);
    return 0;
}

#include <stdio.h>
#define TRUE  1
#define FALSE  0

//compara se o elem está em um vetor
int esta_em(int *v, int n, int x){ 
    for(int i=0; i<n;i++){
        if(*(v+i) == x){
            return TRUE;
        }
    }
    return FALSE;
}



/* Verificar se um número está em um vetor (busca).
   Comparar dois vetores (interseção).
   Garantir que não haja duplicatas no vetor de resultado (unicidade).*/
int intersecao(int *a, int n, int *b, int m, int *c){
    int k=0;
    for(int i = 0; i<n; i++){
        int elem_atual = *(a+i); //inicializando a comparação de a[i]
        if(esta_em(b,m,elem_atual)){ //compara a[n] com b[m] ou melhor a[i] com b[m]
            if(!esta_em(c,k, elem_atual)){
                *(c+k) = elem_atual; //atribui c[k]=a[i]
                k++;
            }
        }
    }
    return k;
}

int remover(int *v, int n, int pos){
    
    if(n<0 || pos>=n){
        return n;
    }
    for(int i=pos; i<n-1; i++){
        *(v+i) = *(v+i+1);
    }
    return n-1;
}


#include <stdio.h>

    int v[5] = {10, 20, 30, 40, 50};
    int n = 5;
    int pos = 1; // Vamos remover o 20

    printf("Antes: ");
    for(int i=0; i<n; i++) printf("%d ", v[i]);

    n = remover(v, n, pos); // Atualizamos n com o novo tamanho

    printf("\nDepois: ");
    for(int i=0; i<n; i++) printf("%d ", v[i]);
    
    // Saída: 10 30 40 50
    return 0;
}

void frequencia(int *v, int n, int *freq, int max_val) {
    // 1. Primeiro, garantimos que o vetor de frequências comece todo zerado
    for (int j = 0; j <= max_val; j++) {
        *(freq + j) = 0;
    }
    // Para cada elemento no vetor v, incrementamos a posição correspondente em freq
    for (int i = 0; i < n; i++) {
        int valor_atual = *(v + i); // Pega o número (ex: 3)
        (*(freq + valor_atual))++;  // Vai no índice 3 de freq e soma +1
    }
}

int palindromo(int *v, int n) {
    // Percorremos apenas até a metade (n / 2)
    for (int i = 0; i < n / 2; i++) {
        // Comparamos o elemento da frente com o seu correspondente atrás
        if (*(v + i) != *(v + (n - 1 - i))) {// O par de i é sempre (n - 1 - i)
            return 0; // Se um par for diferente, não é palíndromo
        }
    }
    return 1; // Se o laço terminar sem retornar 0, é porque todos os pares são iguais
}
